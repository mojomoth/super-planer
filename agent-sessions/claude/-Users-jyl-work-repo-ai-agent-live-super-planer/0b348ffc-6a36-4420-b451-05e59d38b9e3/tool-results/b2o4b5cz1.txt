     7	/** Trim sentence punctuation and an unbalanced closing paren (markdown `[t](url)`, "(see https://x)"). */
     8	export function trimUrl(raw: string): string {
     9	  let u = raw.replace(/[.,;:!?*]+$/, '');
    10	  while (u.endsWith(')') && (u.match(/\(/g)?.length ?? 0) < (u.match(/\)/g)?.length ?? 0)) {
    11	    u = u.slice(0, -1).replace(/[.,;:!?*]+$/, '');
    12	  }
    13	  return u;
    14	}

    16	/** Every distinct http(s) URL in the text, in order of appearance. */
    17	export function extractUrls(text: string): string[] {
    18	  const out: string[] = [];
    19	  for (const m of text.matchAll(URL_RE)) {
    20	    const u = trimUrl(m[0]);
    21	    if (!isHttpUrl(u) || out.includes(u)) continue;
    22	    out.push(u);
    23	  }
    24	  return out;
    25	}

   110	export async function decisionFor(card: Card, history: Message[]): Promise<{ decision: string; provider: Provider; error?: string }> {
   111	  const lastAgent = [...history].reverse().find((m) => m.role === 'agent');
   112	  // prefer the suggestions (list items) over the reply's framing sentences
   113	  const gist = (text: string) => {
   114	    const items = parseBlocks(text).flatMap((b) => (b.t === 'p' ? [] : b.items)).filter((i) => condense(i));
   115	    return condense(items.length ? items.join('\n') : text);
   116	  };
   117	  const fallback = () => ({ decision: (lastAgent && gist(lastAgent.content)) || card.description || card.title });
   118	  if (!history.length) return { decision: fallback().decision, provider: 'mock' };
   119	  const system = `게임 기획 카드 "${card.title}" (${card.description})의 대화를 읽고, 합의된 결정을 한국어 1~3문장으로 요약한다.
   120	결정이 분명하지 않으면 가장 최근 제안을 결정으로 적는다. URL과 지어낸 수치는 넣지 않는다. JSON만 출력한다: {"decision": "..."}`;

    40	/** True for anything that is not a public unicast address (loopback, private, link-local/metadata, CGNAT, …). */
    41	export function isBlockedIp(ip: string): boolean {
    42	  const v = net.isIP(ip);
    43	  if (!v) return true;
    44	  if (v === 4) return blocked4.check(ip, 'ipv4');
    45	  if (embeds4.check(ip, 'ipv6')) {
    46	    // URL normalises to compressed hex, so the last two groups are the low 32 bits ("::" → 0)
    47	    const g = new URL(`http://[${ip}]/`).hostname.slice(1, -1).split(':').slice(-2).map((x) => parseInt(x || '0', 16));
    48	    return isBlockedIp(`${g[0] >> 8}.${g[0] & 255}.${g[1] >> 8}.${g[1] & 255}`);
    49	  }
    50	  return blocked6.check(ip, 'ipv6');
    51	}

    91	function readCapped(res: http.IncomingMessage): Promise<Buffer> {
    92	  const enc = String(res.headers['content-encoding'] ?? '').toLowerCase();
    93	  const stream: Readable =
    94	    enc === 'gzip' ? res.pipe(zlib.createGunzip())
    95	    : enc === 'deflate' ? res.pipe(zlib.createInflate())
    96	    : enc === 'br' ? res.pipe(zlib.createBrotliDecompress())
    97	    : res;
    98	  return new Promise((resolve, reject) => {
    99	    const chunks: Buffer[] = [];
   100	    let size = 0;
   101	    let done = false;
   102	    const finish = () => {
   103	      if (done) return;
   104	      done = true;
   105	      res.destroy();
   106	      resolve(Buffer.concat(chunks).subarray(0, MAX_BYTES));
   107	    };
   108	    stream.on('data', (c: Buffer) => {
   109	      chunks.push(c);
   110	      size += c.length;
   111	      if (size >= MAX_BYTES) finish(); // shortcut: parse only the first 1MB, enough for <head>
   112	    });
   113	    stream.on('end', finish);
   114	    const fail = (e: Error) => {
   115	      if (done) return;
   116	      done = true;
   117	      reject(e);
   118	    };
   119	    stream.on('error', fail);
   120	    if (stream !== res) res.on('error', fail);
   121	  });
   122	}

   183	/** Fetch a page's preview. Throws on a blocked target, a non-2xx, a non-HTML body or a timeout. */
   184	export async function fetchPreview(raw: string): Promise<Preview> {
   185	  let u = new URL(raw);
   186	  const signal = AbortSignal.timeout(TIMEOUT_MS);
   187	  for (let hop = 0; ; hop++) {

   191	    if (status >= 300 && status < 400 && res.headers.location) {
   192	      res.resume();

   194	      u = new URL(res.headers.location, u);
   195	      continue;
   196	    }
   197	    if (status < 200 || status >= 300) {
   198	      res.resume();
   199	      throw new Error(`HTTP ${status}`);
   200	    }
   201	    const type = String(res.headers['content-type'] ?? '');
   202	    if (/^image\//i.test(type)) {
   203	      res.destroy();
   204	      return { url: raw, kind: 'image', title: null, description: null, image: u.href, site: u.hostname };
   205	    }

   207	      res.destroy();
   208	      throw new Error(`not HTML (${type || 'no content-type'})`);
   209	    }
   210	    const html = decode(await readCapped(res), type);
   211	    return { url: raw, kind: 'page', ...parseMeta(html, u) };
   212	  }

    13	const BULLET = /^\s*[-*•]\s+(.*)$/;
    14	const NUMBER = /^\s*\d+[.)]\s+(.*)$/;
    15	
    16	export function parseBlocks(text: string): Block[] {
    17	  const blocks: Block[] = [];
    18	  for (const chunk of text.replace(/\r\n/g, '\n').split(/\n\s*\n/)) {
    19	    const lines = chunk.split('\n').filter((l) => l.trim());
    20	    let para: string[] = [];
    21	    let list: { t: 'ul' | 'ol'; items: string[] } | null = null;
    22	    const flush = () => {
    23	      if (para.length) blocks.push({ t: 'p', lines: para });
    24	      if (list) blocks.push(list);
    25	      para = [];
    26	      list = null;
    27	    };
    28	    for (const line of lines) {
    29	      const b = BULLET.exec(line);
    30	      const n = b ? null : NUMBER.exec(line);
    31	      const kind = b ? 'ul' : n ? 'ol' : null;
    32	      if (kind) {
    33	        if (para.length || (list && list.t !== kind)) flush();
    34	        list ??= { t: kind, items: [] };
    35	        list.items.push((b ?? n)![1]);
    36	      } else {
    37	        if (list) flush();
    38	        para.push(line.trim());
    39	      }
    40	    }
    41	    flush();
    42	  }
    43	  return blocks;
    44	}

    46	const INLINE_RE = /\*\*([^*]+)\*\*|\[([^\]]+)\]\((https?:\/\/[^\s)]+)\)|(https?:\/\/[A-Za-z0-9\-._~:/?#@!$&*+,;=%()]+)/g;

    48	export function parseInline(line: string): Inline[] {
    49	  const out: Inline[] = [];
    50	  let last = 0;
    51	  for (const m of line.matchAll(INLINE_RE)) {
    52	    let end = m.index + m[0].length;
    53	    let tok: Inline;
    54	    if (m[1] !== undefined) tok = { t: 'bold', v: m[1] };
    55	    else if (m[2] !== undefined) tok = { t: 'link', v: m[2], href: m[3] };
    56	    else {
    57	      const u = trimUrl(m[4]); // trailing punctuation / unbalanced ")" stays as text
    58	      end = m.index + u.length;
    59	      tok = { t: 'url', href: u };

    62	    out.push(tok);
    63	    last = end;
    64	  }
    65	  if (last < line.length) out.push({ t: 'text', v: line.slice(last) });

    70	export function condense(text: string, max = 220): string {
    71	  const flat = text
    72	    .replace(/\[([^\]]+)\]\(https?:\/\/[^)]+\)/g, '$1')
    73	    .split('\n')
    74	    .map((l) => l.replace(/https?:\/\/\S+/g, '').replace(/\*\*/g, '').replace(/^\s*(?:[-*•]|\d+[.)])\s+/, '').trim())
    75	    .filter((l) => l && !/^(참고|분위기 참고|비슷한 구조)\s*:?$/.test(l) && !/:\s*$/.test(l))

    82	  return end > max / 3 ? cut.slice(0, end + 1).trim() : `${cut.trimEnd()}…`;
