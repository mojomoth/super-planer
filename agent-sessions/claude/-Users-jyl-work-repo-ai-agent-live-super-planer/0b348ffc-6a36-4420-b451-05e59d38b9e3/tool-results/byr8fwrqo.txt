    49	/** 을/를, 이/가 … picked by the last syllable's final consonant. */
    50	export function josa(word: string, withFinal: string, without: string): string {
    51	  const c = word.replace(/[\s'"’”)\].,!?…]+$/, '').slice(-1).charCodeAt(0);
    52	  if (c >= 0xac00 && c <= 0xd7a3) return (c - 0xac00) % 28 ? withFinal : without;
    53	  return without;
    54	}

    56	const byPosition = (prompt: string, words: string[]) => words.filter((w) => prompt.includes(w)).sort((a, b) => prompt.indexOf(a) - prompt.indexOf(b));
    57	
    58	/** Characters and places in prompt order. `player` is a character written as the subject of something it does
    59	 *  ("고양이가 운영하는"), not one that merely shows up ("좀비가 나오는"), so enemies are never offered as the player. */
    60	export function themes(prompt: string): { chars: string[]; places: string[]; player: string | null } {
    61	  const chars = byPosition(prompt, CHARACTERS);
    62	  const player = chars.find((c) => new RegExp(`${c}(?:이|가|은|는)(?!\\s*(?:나오|나타|등장|쫓아|몰려|습격|공격|덮치))`).test(prompt)) ?? null;
    63	  return { chars: chars.slice(0, 1), places: byPosition(prompt, PLACES).slice(0, 2), player };
    64	}

    66	function subject(prompt: string): string {
    67	  const { chars, places } = themes(prompt);
    68	  const words = [...chars, ...places].slice(0, 2);
    69	  if (!words.length) return '이 게임';
    70	  if (words.length === 1) return `${words[0]}${josa(words[0], '이', '가')} 나오는 이 게임`;
    71	  return `${words[0]}${josa(words[0], '과', '와')} ${words[1]}${josa(words[1], '이', '가')} 나오는 이 게임`;
    72	}

    74	export function initialLedger(prompt: string): Ledger {
    75	  const l = {} as Ledger;
    76	  for (const k of TRACK_KEYS) l[k] = SIGNALS[k].test(prompt) ? 0.55 : 1;
    77	  if (themes(prompt).player) l.role = Math.min(l.role, 0.7);
    78	  return l;
    79	}

    81	export function overall(l: Ledger): number {
    82	  return round(TRACKS.reduce((s, t) => s + t.weight * clamp(l[t.key] ?? 1), 0));
    83	}

    85	/** Deterministic re-score after an answer: the answered track drops by how concrete the answer was, and every further
    86	 *  answer on it drops it at least 0.15 more. Words the player typed (not the preset option labels) lower other tracks a little. */
    87	export function applyAnswer(l: Ledger, track: TrackKey, a: Answer): Ledger {
    88	  const next = { ...l };
    89	  const picked = a.choices.filter((c) => c !== RECOMMEND);
    90	  const text = a.text.trim();
    91	  let score: number;
    92	  if (picked.length && text) score = 0.05;
    93	  else if (picked.length) score = 0.15;
    94	  else if (text.length >= 6) score = 0.1;
    95	  else if (a.choices.includes(RECOMMEND)) score = 0.3;
    96	  else score = 0.45; // a very short free answer stays fuzzy
    97	  next[track] = clamp(Math.min(l[track] - 0.15, score));
    98	  for (const k of TRACK_KEYS) if (k !== track && SIGNALS[k].test(text)) next[k] = Math.min(next[k], 0.5);
    99	  return next;
   100	}

   102	export const blockedTrack = (history: TrackKey[]): TrackKey | null =>
   103	  history.length >= 2 && history.at(-1) === history.at(-2) ? history.at(-1)! : null;
   104	
   105	/** Most ambiguous track; ties go to the track asked fewer times, then the heavier one.
   106	 *  After 2 consecutive turns on one track, zoom out to another (ouroboros breadth control).
   107	 *  The first question is the ontological one ("what IS this game?") unless the prompt already pinned the fun down. */
   108	export function pickTrack(l: Ledger, history: TrackKey[]): TrackKey {
   109	  if (!history.length && l.fun > CLEAR) return 'fun';
   110	  const blocked = blockedTrack(history);
   111	  const asks = (k: TrackKey) => history.filter((h) => h === k).length;
   112	  return TRACKS.filter((t) => t.key !== blocked)
   113	    .sort((a, b) => l[b.key] - l[a.key] || asks(a.key) - asks(b.key) || b.weight - a.weight)[0].key;
   114	}
   115	
   116	export function shouldStop(l: Ledger, asked: number): boolean {
   117	  return overall(l) <= THRESHOLD || asked >= MAX_QUESTIONS;
   118	}

   120	/** One readable decision per answered turn ("추천" picks the first option, which is always the recommended one). */
   121	export function decisionOf(t: Turn): string {
   122	  if (!t.answer) return '';
   123	  const parts = t.answer.choices.map((c) => (c === RECOMMEND ? `${t.options[0] ?? '추천안'}(추천)` : c));
   124	  if (t.answer.text.trim()) parts.push(t.answer.text.trim());
   125	  return parts.join(', ');
   126	}

   128	export function decisions(turns: Turn[]): Decision[] {
   129	  const out: Decision[] = [];
   130	  for (const k of TRACK_KEYS) {
   131	    const d = turns.filter((t) => t.track === k && t.answer).map(decisionOf).filter(Boolean);
   132	    if (d.length) out.push({ key: k, track: label(k), decision: d.join(', ') });
   133	  }
   134	  return out;
   135	}

   137	const short = (s: string, n = 24) => (s.length > n ? `${s.slice(0, n).trim()}…` : s);
   138	/** What was said to a track's main (first) question: picked options ("추천" resolved to the first option) plus the typed text.
   139	 *  Follow-up answers dig deeper and live in the decisions table. */
   140	function picks(turns: Turn[], k: TrackKey): string[] {
   141	  const t = turns.find((x) => x.track === k && x.answer);
   142	  if (!t) return [];
   143	  return [...t.answer!.choices.map((c) => (c === RECOMMEND ? t.options[0] ?? '' : c)), t.answer!.text.trim()].filter(Boolean).map((s) => short(s));
   144	}
   145	const said = (turns: Turn[], k: TrackKey) => picks(turns, k)[0] ?? '';

   150	  ['survive', ['버티', '살아남', '생존', '좀비', '탈출', '하룻밤', '도망', '숨어']],

   152	  ['puzzle', ['퍼즐', '머리', '푸는', '추리', '탐정', '두뇌', '맞추']],
   153	  ['action', ['액션', '싸우', '전투', '피하', '슈팅', '달리', '점프', '빠르', '짜릿', '리듬']],
   154	  ['story', ['이야기', '스토리', '소설', '선택지', '연애', '대화', '기억', '비밀', '추억']],
   155	  ['versus', ['대전', '겨루', 'pvp', '배틀', '경쟁', '대결', '멀티']],
   156	];
   157	export function kind(text: string): Kind {
   158	  const t = text.toLowerCase();
   159	  let best: Kind = 'other';
   160	  let hits = 0;
   161	  for (const [k, words] of KINDS) {
   162	    const n = words.filter((w) => t.includes(w)).length;
   163	    if (n > hits) [best, hits] = [k, n];
   164	  }
   165	  return best;
   166	}

   193	/** Put options that match the text first (the first option doubles as the recommendation). */
   194	function favour(options: string[], text: string, rules: [RegExp, string][]): string[] {
   195	  const hit = rules.filter(([re]) => re.test(text)).map(([, o]) => o);

   328	/** Restate gate goal line, and the mock concept for the result screen (ELI5: what, analogy, details, so what). */
   329	export function goalLine(prompt: string, turns: Turn[]): string {
   330	  const role = said(turns, 'role').replace(/\(.*?\)/g, '').trim();
   331	  const loop = said(turns, 'loop');
   332	  if (role && loop) return `${role}${josa(role, '이', '가')} 되어 ${loop}${josa(loop, '을', '를')} 반복하는 게임이에요.`;
   333	  if (loop) return `${loop}${josa(loop, '을', '를')} 반복하는 게임이에요.`;
   334	  return `“${short(prompt, 60)}”를 바탕으로 한 게임이에요.`;
   335	}

   348	export function mockConcept(prompt: string, turns: Turn[]): Concept {
   349	  const k = gameKind(prompt, turns);
   350	  const { chars, places } = themes(prompt);
   351	  const head = [...places.slice(0, 1), ...chars].join(' ') || '나만의';
   352	  const fun = said(turns, 'fun');
   353	  const loop = said(turns, 'loop');
   354	  const audience = said(turns, 'audience');
   355	  const platform = said(turns, 'platform');
   356	  const unique = said(turns, 'unique');
   357	  const role = said(turns, 'role');
   358	  const loops = picks(turns, 'loop');
   359	  const win = said(turns, 'win');
   360	  const moods = picks(turns, 'mood');
   361	  // "조금 더 자세히": how a session plays out, not the decisions table again.
   362	  const details = [
   363	    role && loops[0] && `처음 켜면 ‘${role}’ 입장에서 바로 ‘${loops[0]}’부터 해요.`,
   364	    loops.length > 1 ? `한 바퀴(계속 반복하는 흐름)는 ${loops.slice(0, 4).join(' → ')} → 다시 처음이에요.` : loops[0] && `한 바퀴(계속 반복하는 흐름)는 ‘${loops[0]}’ 하나로 단순하게 가요.`,
   365	    win && (/끝이 없/.test(win) ? '정해진 끝 없이 계속 이어지고, 그만큼 내 공간이 자라요.' : `한 판은 ‘${win}’ 규칙으로 끝나요.`),
   366	    platform && `‘${platform}’에 맞춰 한 판 길이와 조작을 짜요.`,
   367	    moods.length > 0 && `화면과 소리는 ${moods.slice(0, 3).map((m) => `‘${m}’`).join(', ')} 쪽으로 맞춰요.`,
   368	  ].filter((d): d is string => !!d);
   369	  return {
   370	    title: `${head} ${TITLE_TAIL[k]}`,
   371	    one_liner: goalLine(prompt, turns),
   372	    what: fun && loop ? `한마디로 ${fun}${josa(fun, '으로', '로')} ${loop}${josa(loop, '을', '를')} 계속하는 게임이에요.` : `한마디로 “${short(prompt, 40)}”를 직접 해 보는 게임이에요.`,
   373	    analogy: ANALOGY[k],
   374	    details: details.length ? details : [`처음 아이디어 “${short(prompt, 40)}”에서 출발해요.`, '질문에 더 답할수록 이 부분이 구체적으로 채워져요.'],
   375	    so_what: [
   376	      audience ? `‘${audience}’도 ${platform ? `‘${platform}’ 동안` : '부담 없이'} 바로 즐길 수 있어요.` : '처음 하는 사람도 바로 알아볼 수 있어요.',
   377	      unique ? `‘${unique}’ 덕분에 비슷한 게임 사이에서 기억에 남아요.` : '한 판이 끝나면 다음 판이 궁금해지게 만드는 게 목표예요.',
   378	    ].join(' '),
   379	  };

   420	export function parseQuestion(raw: unknown, blocked: TrackKey | null): { q: Question | null; ledger: Partial<Ledger>; done: boolean } {
   421	  const r = (raw ?? {}) as Record<string, unknown>;
   422	  const ledger: Partial<Ledger> = {};
   423	  const amb = (r.ambiguity ?? {}) as Record<string, unknown>;
   424	  for (const k of TRACK_KEYS) if (typeof amb[k] === 'number' && Number.isFinite(amb[k])) ledger[k] = clamp(amb[k] as number);
   425	  const done = r.done === true;
   426	  const track = r.track as TrackKey;
   427	  const question = str(r.question);
   428	  const options = Array.isArray(r.options) ? [...new Set(r.options.map((o) => str(o, 80)).filter((o) => o && o !== RECOMMEND))].slice(0, 5) : [];
   429	  const ok = TRACK_KEYS.includes(track) && track !== blocked && /[?？]$/.test(question) && options.length >= 2;
   430	  return { q: ok ? { track, question, hint: str(r.hint, 200), options, multi: r.multi === true, allow_free: r.allow_free !== false } : null, ledger, done };
   431	}

   445	const LABEL = /^(이게 뭐냐면|비유하자면|조금 더 자세히|그래서 왜 재밌냐면)\s*[:：]\s*/;
   446	export function parseConcept(raw: unknown, fallback: Concept): Concept {
   447	  const r = (raw ?? {}) as Record<string, unknown>;
   448	  const str = (v: unknown, max = 300) => (typeof v === 'string' ? v.trim().replace(LABEL, '').slice(0, max) : '');
   449	  const details = Array.isArray(r.details) ? r.details.map((d) => str(d, 200)).filter(Boolean).slice(0, 8) : [];
   450	  return {
   451	    title: str(r.title, 40) || fallback.title,
   452	    one_liner: str(r.one_liner) || fallback.one_liner,
   453	    what: str(r.what) || fallback.what,
   454	    analogy: str(r.analogy) || fallback.analogy,
   455	    details: details.length ? details : fallback.details,
   456	    so_what: str(r.so_what) || fallback.so_what,
   457	  };
   458	}
