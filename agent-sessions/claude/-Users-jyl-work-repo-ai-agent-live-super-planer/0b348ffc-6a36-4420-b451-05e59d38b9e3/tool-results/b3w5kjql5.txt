    25	async function api<T>(url: string, init?: RequestInit): Promise<T> {
    26	  const res = await fetch(url, init);
    27	  const data = (await res.json().catch(() => ({}))) as T & { error?: string; message?: string };
    28	  if (!res.ok) throw new Error(data.error ?? data.message ?? `요청이 실패했어요 (${res.status})`);
    29	  return data;
    30	}
    31	const json = (method: string, body: unknown): RequestInit => ({ method, headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) });
    32	
    33	/** One-line text preview of a Markdown body. */
    34	const preview = (body: string | null) => (body ?? '').replace(/[#*_`>|-]+/g, ' ').replace(/\s+/g, ' ').trim().slice(0, 140);
    35	const when = (iso: string) => new Date(iso).toLocaleString('ko-KR', { month: 'numeric', day: 'numeric', hour: '2-digit', minute: '2-digit' });

    41	  const [docs, setDocs] = useState(initialDocs);
    42	  const [doc, setDoc] = useState<DocRow | null>(initialDoc);
    43	  const [md, setMd] = useState(initialDoc?.markdown ?? '');
    44	  const [save, setSave] = useState<Save>('saved');
    45	  const [tab, setTab] = useState<Tab>('preview');

    47	  const [error, setError] = useState<string | null>(null);
    48	  const [note, setNote] = useState<string | null>(null);
    49	  const [copied, setCopied] = useState(false);
    50	  const [editing, setEditing] = useState<InputRow | 'new' | null>(null);
    51	  const [captioning, setCaptioning] = useState<string | null>(null);
    52	  const [addingImage, setAddingImage] = useState(false);
    53	  const [step1Open, setStep1Open] = useState(false);
    54	  const [step1, setStep1] = useState(initialStep1);
    55	  const tabRefs = useRef<Record<Tab, HTMLButtonElement | null>>({ preview: null, markdown: null, split: null });
    56	
    57	  const decisions = inputs.filter((i) => i.kind === 'decision');
    58	  const images = inputs.filter((i) => i.kind === 'image');
    59	  const pickedDecisions = decisions.filter((i) => selected.has(i.id));
    60	  const pickedImages = images.filter((i) => selected.has(i.id));
    61	
    62	  const importedCount = imported?.ids.length
    63	    ? `워크플로우에서 결정 ${inputs.filter((i) => imported.ids.includes(i.id) && i.kind === 'decision').length}개, 이미지 ${inputs.filter((i) => imported.ids.includes(i.id) && i.kind === 'image').length}개를 가져와 골라 두었어요.`
    64	    : null;
    65	
    66	  // /?workflow=<id> hand-off: import once (idempotent, keeps edited copies), preselect, then drop the query so a refresh does not repeat it.
    67	  useEffect(() => {
    68	    if (!workflowId) return;
    69	    let live = true;
    70	    (async () => {
    71	      try {
    72	        const res = await fetch('/api/import', json('POST', { workflow: workflowId }));
    73	        const data = (await res.json().catch(() => ({}))) as Partial<Imported> & { inputs?: InputRow[]; error?: string };
    74	        if (!live) return;
    75	        if (data.inputs) setInputs(data.inputs);
    76	        if (data.ids?.length) setSelected(new Set(data.ids));
    77	        setImported({ ids: data.ids ?? [], message: data.ids?.length ? undefined : (data.message ?? data.error ?? '워크플로우를 가져오지 못했어요.') });
    78	        window.history.replaceState(null, '', window.location.pathname);
    79	      } catch {
    80	        if (live) setImported({ ids: [], message: '워크플로우를 가져오지 못했어요. 새로고침해 다시 시도해 주세요.' });
    81	      }
    82	    })();
    83	    return () => {
    84	      live = false;
    85	    };
    86	  }, [workflowId]);
    87	
    88	  // Debounced autosave of the Markdown editor.
    89	  useEffect(() => {
    90	    if (save !== 'dirty' || !doc) return;
    91	    const t = setTimeout(() => void persist(doc.id, md), 800);
    92	    return () => clearTimeout(t);
    93	  }, [md, save, doc]);
    94	
    95	  async function persist(id: string, markdown: string) {
    96	    setSave('saving');
    97	    try {
    98	      const r = await api<{ doc: DocRow }>(`/api/docs/${id}`, json('PATCH', { markdown }));
    99	      setDocs((ds) => ds.map((d) => (d.id === id ? { ...d, title: r.doc.title } : d)));
   100	      setDoc((d) => (d && d.id === id ? { ...d, title: r.doc.title } : d));
   101	      setSave((s) => (s === 'saving' ? 'saved' : s));
   102	    } catch {
   103	      setSave('error');
   104	      setTimeout(() => setSave((s) => (s === 'error' ? 'dirty' : s)), 3000);
   105	    }
   106	  }
   107	
   108	  function toggle(id: string) {
   109	    setSelected((s) => {
   110	      const n = new Set(s);
   111	      if (n.has(id)) n.delete(id);
   112	      else n.add(id);
   113	      return n;
   114	    });
   115	  }
   116	
   117	  function setAll(list: InputRow[], on: boolean) {
   118	    setSelected((s) => {
   119	      const n = new Set(s);
   120	      for (const i of list) if (on) n.add(i.id);
   121	      else n.delete(i.id);
   122	      return n;
   123	    });
   124	  }
   125	
   126	  async function generate() {
   127	    setBusy(true);
   128	    setError(null);
   129	    setNote(null);
   130	    try {
   131	      if (doc && save !== 'saved') await persist(doc.id, md);
   132	      const r = await api<{ doc: DocRow; error?: string }>('/api/generate', json('POST', { decisionIds: pickedDecisions.map((i) => i.id), imageIds: pickedImages.map((i) => i.id) }));
   133	      setDoc(r.doc);
   134	      setMd(r.doc.markdown);
   135	      setSave('saved');
   136	      setDocs((ds) => [{ id: r.doc.id, title: r.doc.title, provider: r.doc.provider, created_at: r.doc.created_at }, ...ds]);
   137	      setTab('preview');
   138	      if (r.error) setNote(`모델 응답을 쓰지 못해 입력만으로 조립한 문서를 만들었어요. (${r.error})`);
   139	    } catch (e) {
   140	      setError((e as Error).message);
   141	    } finally {
   142	      setBusy(false);
   143	    }
   144	  }
   145	
   146	  async function openDoc(id: string) {
   147	    if (id === doc?.id) return;
   148	    setError(null);
   149	    try {
   150	      if (doc && save !== 'saved') await persist(doc.id, md);
   151	      const r = await api<{ doc: DocRow }>(`/api/docs/${id}`);
   152	      setDoc(r.doc);
   153	      setMd(r.doc.markdown);
   154	      setSave('saved');
   155	    } catch (e) {
   156	      setError((e as Error).message);
   157	    }
   158	  }
   159	
   160	  async function removeInput(i: InputRow) {
   161	    if (!window.confirm(`'${i.title}'을(를) 지울까요?`)) return;
   162	    try {
   163	      await api(`/api/inputs/${encodeURIComponent(i.id)}`, { method: 'DELETE' });
   164	      setInputs((xs) => xs.filter((x) => x.id !== i.id));
   165	      if (editing !== 'new' && editing?.id === i.id) setEditing(null);
   166	    } catch (e) {
   167	      setError((e as Error).message);
   168	    }
   169	  }
   170	
   171	  function added(input: InputRow) {
   172	    setInputs((xs) => [...xs.filter((x) => x.id !== input.id), input]);
   173	    setSelected((s) => new Set(s).add(input.id));
   174	  }
   175	
   176	  async function loadSeed() {
   177	    try {
   178	      const r = await api<{ inputs: InputRow[] }>('/api/seed', { method: 'POST' });
   179	      setInputs(r.inputs);
   180	      setSelected(new Set(r.inputs.map((i) => i.id)));
   181	    } catch (e) {
   182	      setError((e as Error).message);
   183	    }
   184	  }
   185	
   186	  async function openStep1() {
   187	    setStep1Open((o) => !o);
   188	    if (step1Open) return;
   189	    try {
   190	      setStep1((await api<{ step1: Step1Row[] }>('/api/import')).step1);
   191	    } catch {
   192	      // keep the list we had from the server render
   193	    }
   194	  }
   195	
   196	  async function importStep1(id: string) {
   197	    setError(null);
   198	    try {
   199	      const r = await api<{ ids: string[]; inputs: InputRow[] }>('/api/import', json('POST', { step1: id }));
   200	      setInputs(r.inputs);
   201	      setSelected((s) => new Set([...s, ...r.ids]));
   202	      setStep1Open(false);
   203	    } catch (e) {
   204	      setError((e as Error).message);
   205	    }
   206	  }
   207	
   208	  function onTabKey(e: KeyboardEvent<HTMLButtonElement>) {
   209	    const i = TABS.findIndex(([t]) => t === tab);
   210	    const next = e.key === 'ArrowRight' ? (i + 1) % TABS.length : e.key === 'ArrowLeft' ? (i + TABS.length - 1) % TABS.length : e.key === 'Home' ? 0 : e.key === 'End' ? TABS.length - 1 : -1;
   211	    if (next < 0) return;
   212	    e.preventDefault();
   213	    setTab(TABS[next][0]);
   214	    tabRefs.current[TABS[next][0]]?.focus();
   215	  }
   216	
   217	  // Relative image paths (/samples, /api/images) only resolve inside this app, so exports get absolute URLs.
   218	  const exportMd = () => md.replace(/\]\(\/(?!\/)/g, `](${window.location.origin}/`);
   219	
   220	  function download() {
   221	    const blob = new Blob([exportMd()], { type: 'text/markdown;charset=utf-8' });
   222	    const a = document.createElement('a');
   223	    a.href = URL.createObjectURL(blob);
   224	    a.download = `${(doc?.title ?? '기획서').replace(/[\\/:*?"<>|]+/g, ' ').trim() || '기획서'} 기획서.md`;
   225	    a.click();
   226	    // Revoking right after click() can cancel the download in some browsers.
   227	    setTimeout(() => URL.revokeObjectURL(a.href), 1000);
   228	  }
   229	
   230	  async function copy() {
   231	    try {
   232	      await navigator.clipboard.writeText(exportMd());
   233	      setCopied(true);
   234	      setTimeout(() => setCopied(false), 2000);
   235	    } catch {
   236	      setError('클립보드에 복사하지 못했어요. 마크다운 탭에서 직접 선택해 복사해 주세요.');
   237	    }
   238	  }
   239	
   240	  const editor = (
   241	    <div className="field doc-editor">
   242	      <label className="field__label" htmlFor="doc-md">마크다운</label>
   243	      <textarea
   244	        id="doc-md"
   245	        className="textarea textarea--code doc-editor__area"
   246	        value={md}
   247	        spellCheck={false}
   248	        onChange={(e) => {
   249	          setMd(e.target.value);
   250	          setSave('dirty');
   251	        }}
   252	      />
   253	    </div>
   254	  );
   255	
   256	  return (
   257	    <>
   258	      <div className="page-head">
   259	        <div className="page-head__text">
   260	          <h1>기획서 만들기</h1>
   261	          <p className="page-head__desc">결정 문서와 이미지 예시를 골라 한 편의 Markdown 기획서로 묶어요.</p>
   262	        </div>
   263	      </div>
   264	
   265	      {imported?.message ? (
   266	        <div className="notice plan-notice" role="status"><Icon name="alert" /><p>{imported.message}</p></div>
   267	      ) : importedCount ? (
   268	        <div className="notice plan-notice" role="status"><Icon name="check" /><p>{importedCount}</p></div>
   269	      ) : null}
   270	      {error ? (
   271	        <div className="notice notice--error plan-notice" role="alert"><Icon name="alert" /><p>{error}</p></div>
   272	      ) : null}
   273	
   274	      <div className="plan-layout">
   275	        <div className="plan-inputs">
   276	          <section className="panel" aria-labelledby="dec-h">
   277	            <header className="panel__head">
   278	              <div className="panel__heading">
   279	                <h2 className="panel__title" id="dec-h">결정 문서</h2>
   280	                <p className="meta num">{pickedDecisions.length} / {decisions.length}개 선택</p>
   281	              </div>
   282	              <div className="panel__actions">
   283	                <button type="button" className="btn btn--ghost btn--sm" onClick={() => setAll(decisions, pickedDecisions.length !== decisions.length)} disabled={!decisions.length}>
   284	                  {pickedDecisions.length === decisions.length && decisions.length ? '모두 해제' : '모두 선택'}
   285	                </button>
   286	                <button type="button" className="btn btn--sm" onClick={() => setEditing('new')} aria-expanded={editing === 'new'}>
   287	                  <Icon name="plus" />추가
   288	                </button>
   289	              </div>
   290	            </header>
   291	            <div className="panel__body">
   292	              {editing ? (
   293	                <DecisionForm
   294	                  key={editing === 'new' ? 'new' : editing.id}
   295	                  input={editing === 'new' ? null : editing}
   296	                  onDone={(row) => {
   297	                    if (row) {
   298	                      if (editing === 'new') added(row);
   299	                      else setInputs((xs) => xs.map((x) => (x.id === row.id ? row : x)));
   300	                    }
   301	                    setEditing(null);
   302	                  }}
   303	                />
   304	              ) : null}
   305	              {decisions.length ? (
   306	                <ul className="pick-list">
   307	                  {decisions.map((d) => (
   308	                    <li key={d.id} className="pick">
   309	                      <input type="checkbox" id={`pick-${d.id}`} checked={selected.has(d.id)} onChange={() => toggle(d.id)} />
   310	                      <label htmlFor={`pick-${d.id}`} className="pick__text">
   311	                        <span className="pick__title">{d.title}</span>
   312	                        <span className="pick__preview">{preview(d.body)}</span>
   313	                      </label>
   314	                      <div className="pick__actions">
   315	                        <span className="pick__source">{sourceLabel(d.source)}</span>
   316	                        <button type="button" className="btn btn--ghost btn--sm" onClick={() => setEditing(d)} aria-label={`${d.title} 수정`}>수정</button>
   317	                        <button type="button" className="btn btn--ghost btn--danger btn--sm" onClick={() => void removeInput(d)} aria-label={`${d.title} 삭제`}>삭제</button>
   318	                      </div>
   319	                    </li>
   320	                  ))}
   321	                </ul>
   322	              ) : (
   323	                <div className="empty">
   324	                  <p>결정 문서가 없어요. 직접 추가하거나 예시를 불러오세요.</p>
   325	                  <button type="button" className="btn" onClick={() => void loadSeed()}>예시 불러오기</button>
   326	                </div>
   327	              )}
   328	            </div>
   329	            <footer className="panel__foot plan-step1">
   330	              <button type="button" className="btn btn--ghost btn--sm" onClick={() => void openStep1()} aria-expanded={step1Open} aria-controls="step1-list">
   331	                Step1 결과 가져오기
   332	              </button>
   333	              {step1Open ? (
   334	                <div id="step1-list" className="plan-step1__list">
   335	                  {step1.length ? (
   336	                    <ul className="pick-list">
   337	                      {step1.map((s) => (
   338	                        <li key={s.session_id} className="pick pick--plain">
   339	                          <span className="pick__text">
   340	                            <span className="pick__title">{s.title}</span>
   341	                            <span className="pick__preview">{s.one_liner}</span>
   342	                          </span>
   343	                          <button type="button" className="btn btn--sm" onClick={() => void importStep1(s.session_id)}>가져오기</button>
   344	                        </li>
   345	                      ))}
   346	                    </ul>
   347	                  ) : (
   348	                    <p className="meta">
   349	                      Step1에서 끝낸 결과가 아직 없어요. <a href="http://localhost:3101/">질문하기에서 시작하기</a>
   350	                    </p>
   351	                  )}
   352	                </div>
   353	              ) : null}
   354	            </footer>
   355	          </section>
   356	
   357	          <section className="panel" aria-labelledby="img-h">
   358	            <header className="panel__head">
   359	              <div className="panel__heading">
   360	                <h2 className="panel__title" id="img-h">이미지 예시</h2>
   361	                <p className="meta num">{pickedImages.length} / {images.length}개 선택</p>
   362	              </div>
   363	              <div className="panel__actions">
   364	                <button type="button" className="btn btn--ghost btn--sm" onClick={() => setAll(images, pickedImages.length !== images.length)} disabled={!images.length}>
   365	                  {pickedImages.length === images.length && images.length ? '모두 해제' : '모두 선택'}
   366	                </button>
   367	                <button type="button" className="btn btn--sm" onClick={() => setAddingImage((v) => !v)} aria-expanded={addingImage}>
   368	                  <Icon name="plus" />추가
   369	                </button>
   370	              </div>
   371	            </header>
   372	            <div className="panel__body">
   373	              {addingImage ? (
   374	                <ImageForm
   375	                  onDone={(row) => {
   376	                    if (row) added(row);
   377	                    setAddingImage(false);
   378	                  }}
   379	                />
   380	              ) : null}
   381	              {images.length ? (
   382	                <section className="gallery" aria-label="이미지 예시 목록">
   383	                  <ul className="gallery__track" data-view="grid">
   384	                    {images.map((img) => (
   385	                      <li key={img.id} className="preview tile">
   386	                        <input type="checkbox" className="tile__check" id={`pick-${img.id}`} checked={selected.has(img.id)} onChange={() => toggle(img.id)} />
   387	                        <label htmlFor={`pick-${img.id}`} className="tile__label">
   388	                          <span className="preview__media">
   389	                            <img src={img.url ?? ''} alt="" loading="lazy" width="640" height="400" />
   390	                          </span>
   391	                          {captioning === img.id ? null : <span className="preview__title">{img.title}</span>}
   392	                        </label>
   393	                        {captioning === img.id ? (
   394	                          <CaptionForm
   395	                            input={img}
   396	                            onDone={(row) => {
   397	                              if (row) setInputs((xs) => xs.map((x) => (x.id === row.id ? row : x)));
   398	                              setCaptioning(null);
   399	                            }}
   400	                          />
   401	                        ) : null}
   402	                        <span className="tile__foot">
   403	                          <span className="preview__site">{sourceLabel(img.source)}</span>
   404	                          <span className="tile__btns">
   405	                            <button type="button" className="btn btn--ghost btn--sm" onClick={() => setCaptioning(img.id)} aria-label={`${img.title} 캡션 수정`} aria-expanded={captioning === img.id}>캡션</button>
   406	                            <button type="button" className="btn btn--ghost btn--danger btn--sm tile__del" onClick={() => void removeInput(img)} aria-label={`${img.title} 삭제`}>삭제</button>
   407	                          </span>
   408	                        </span>
   409	                      </li>
   410	                    ))}
   411	                  </ul>
   412	                </section>
   413	              ) : (
   414	                <div className="empty"><p>이미지가 없어요. 파일을 올리거나 https 주소로 추가하세요.</p></div>

   416	            </div>
   417	          </section>
   418	
   419	          <div className="plan-go">
   420	            <button type="button" className="btn btn--primary" onClick={() => void generate()} disabled={busy || !pickedDecisions.length} aria-busy={busy}>
   421	              <Icon name="doc" />기획서 만들기
   422	            </button>
   423	            <p className="meta">{pickedDecisions.length ? `결정 ${pickedDecisions.length}개와 이미지 ${pickedImages.length}개로 만들어요.` : '결정 문서를 하나 이상 골라 주세요.'}</p>
   424	          </div>
   425	        </div>
   426	
   427	        <section className="panel doc-panel" aria-labelledby="doc-h" aria-busy={busy}>
   428	          <header className="panel__head doc-panel__head">
   429	            <div className="doc-panel__title">
   430	              <h2 className="panel__title" id="doc-h">{doc ? `${doc.title} 기획서` : '기획서'}</h2>
