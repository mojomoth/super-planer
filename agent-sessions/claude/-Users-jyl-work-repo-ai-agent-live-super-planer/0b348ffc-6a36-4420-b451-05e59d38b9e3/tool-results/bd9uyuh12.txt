    49	/** 을/를, 이/가 … picked by the last syllable's final consonant. */
    50	export function josa(word: string, withFinal: string, without: string): string {
    51	  const c = word.replace(/[\s'"’”)\].,!?…]+$/, '').slice(-1).charCodeAt(0);
    52	  if (c >= 0xac00 && c <= 0xd7a3) return (c - 0xac00) % 28 ? withFinal : without;
    53	  return without;
    54	}

    56	export function themes(prompt: string): { chars: string[]; places: string[] } {
    57	  return { chars: CHARACTERS.filter((w) => prompt.includes(w)).slice(0, 1), places: PLACES.filter((w) => prompt.includes(w)).slice(0, 2) };
    58	}

    60	function subject(prompt: string): string {
    61	  const { chars, places } = themes(prompt);
    62	  const words = [...chars, ...places].slice(0, 2);
    63	  if (!words.length) return '이 게임';
    64	  if (words.length === 1) return `${words[0]}${josa(words[0], '이', '가')} 나오는 이 게임`;
    65	  return `${words[0]}${josa(words[0], '과', '와')} ${words[1]}${josa(words[1], '이', '가')} 나오는 이 게임`;
    66	}

    68	export function initialLedger(prompt: string): Ledger {
    69	  const l = {} as Ledger;
    70	  for (const k of TRACK_KEYS) l[k] = SIGNALS[k].test(prompt) ? 0.55 : 1;
    71	  if (themes(prompt).chars.length) l.role = Math.min(l.role, 0.7);
    72	  return l;

    79	/** Deterministic re-score after an answer: the answered track drops by how concrete the answer was; words in it lower other tracks a little. */
    80	export function applyAnswer(l: Ledger, track: TrackKey, a: Answer): Ledger {
    81	  const next = { ...l };
    82	  const picked = a.choices.filter((c) => c !== RECOMMEND);
    83	  const text = a.text.trim();
    84	  let score: number;
    85	  if (picked.length && text) score = 0.05;
    86	  else if (picked.length) score = 0.15;
    87	  else if (text.length >= 6) score = 0.1;
    88	  else if (a.choices.includes(RECOMMEND)) score = 0.3;
    89	  else score = 0.45; // a very short free answer stays fuzzy
    90	  next[track] = Math.min(l[track], score);
    91	  const said = [...picked, text].join(' ');
    92	  for (const k of TRACK_KEYS) if (k !== track && SIGNALS[k].test(said)) next[k] = Math.min(next[k], 0.5);
    93	  return next;
    94	}

    96	export const blockedTrack = (history: TrackKey[]): TrackKey | null =>
    97	  history.length >= 2 && history.at(-1) === history.at(-2) ? history.at(-1)! : null;

    99	/** Most ambiguous track; after 2 consecutive turns on one track, zoom out to another (ouroboros breadth control).
   100	 *  The first question is the ontological one ("what IS this game?") unless the prompt already pinned the fun down. */
   101	export function pickTrack(l: Ledger, history: TrackKey[]): TrackKey {
   102	  if (!history.length && l.fun > CLEAR) return 'fun';
   103	  const blocked = blockedTrack(history);
   104	  let best: TrackKey | null = null;
   105	  for (const k of TRACK_KEYS) if (k !== blocked && (best === null || l[k] > l[best])) best = k;
   106	  return best!;
   107	}

   109	export function shouldStop(l: Ledger, asked: number): boolean {
   110	  return overall(l) <= THRESHOLD || asked >= MAX_QUESTIONS;
   111	}

   113	/** One readable decision per answered turn ("추천" picks the first option, which is always the recommended one). */
   114	export function decisionOf(t: Turn): string {
   115	  if (!t.answer) return '';
   116	  const parts = t.answer.choices.map((c) => (c === RECOMMEND ? `${t.options[0] ?? '추천안'}(추천)` : c));
   117	  if (t.answer.text.trim()) parts.push(t.answer.text.trim());
   118	  return parts.join(', ');
   119	}

   121	export function decisions(turns: Turn[]): Decision[] {
   122	  const out: Decision[] = [];
   123	  for (const k of TRACK_KEYS) {
   124	    const d = turns.filter((t) => t.track === k && t.answer).map(decisionOf).filter(Boolean);
   125	    if (d.length) out.push({ key: k, track: label(k), decision: d.join(', ') });
   126	  }
   127	  return out;
   128	}

   131	function said(turns: Turn[], k: TrackKey): string {

   326	export function parseQuestion(raw: unknown, blocked: TrackKey | null): { q: Question | null; ledger: Partial<Ledger>; done: boolean } {
   327	  const r = (raw ?? {}) as Record<string, unknown>;
   328	  const ledger: Partial<Ledger> = {};
   329	  const amb = (r.ambiguity ?? {}) as Record<string, unknown>;
   330	  for (const k of TRACK_KEYS) if (typeof amb[k] === 'number' && Number.isFinite(amb[k])) ledger[k] = clamp(amb[k] as number);
   331	  const done = r.done === true;
   332	  const track = r.track as TrackKey;
   333	  const question = str(r.question);
   334	  const options = Array.isArray(r.options) ? [...new Set(r.options.map((o) => str(o, 80)).filter((o) => o && o !== RECOMMEND))].slice(0, 5) : [];
   335	  const ok = TRACK_KEYS.includes(track) && track !== blocked && /[?？]$/.test(question) && options.length >= 2;
   336	  return { q: ok ? { track, question, hint: str(r.hint, 200), options, multi: r.multi === true, allow_free: r.allow_free !== false } : null, ledger, done };
   337	}

    70	/** Ask the next question (LLM, or the mock bank when the model is off/fails/breaks a rule). No-op if one is already open. */
    71	async function askNext(id: string, force = false): Promise<void> {
    72	  const x = rows(id)!;
    73	  if (x.t.some((r) => !r.answer) || x.s.status !== 'interviewing') return;
    74	  const turns = x.t.map(toTurn);
    75	  if (turns.length >= HARD_CAP || (!force && shouldStop(x.ledger, turns.length))) return;
    76	  const history = turns.map((t) => t.track);
    77	  const blocked = blockedTrack(history);
    78	  const target = pickTrack(x.ledger, history);
    79	  const mock = () => ({ ...mockQuestion(x.s.prompt, turns, target), ambiguity: x.ledger, done: false });
    80	  const res = await completeJSON(QUESTION_SYSTEM, questionUser(x.s.prompt, turns, x.ledger, target, blocked), mock);
    81	  const parsed = parseQuestion(res.data, blocked);
    82	  const ledger = { ...x.ledger, ...parsed.ledger };
    83	  let q: Question;
    84	  let used: string = res.provider;
    85	  if (parsed.q) q = parsed.q;
    86	  else if (parsed.done && !force && res.provider !== 'mock') {
    87	    db().prepare('UPDATE step1_sessions SET tracks = ?, ambiguity = ? WHERE id = ?').run(JSON.stringify(ledger), overall(ledger), id);
    88	    return; // model says every track is clear: go to the Restate gate
    89	  } else {
    90	    const pick = pickTrack(ledger, history);
    91	    q = mockQuestion(x.s.prompt, turns, pick);
    92	    used = res.provider === 'mock' ? 'mock' : 'mock (LLM 응답을 쓸 수 없어 기본 질문)';
    93	  }
    94	  if (res.error) used = 'mock (LLM 호출 실패)';
    95	  const d = db();
    96	  if (d.prepare('SELECT 1 FROM step1_turns WHERE session_id = ? AND answer IS NULL').get(id)) return; // a parallel request won
    97	  d.prepare('INSERT INTO step1_turns (id, session_id, idx, track, question, hint, options, allow_free, multi, answer, ledger, provider, created_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, NULL, ?, ?, ?)')
    98	    .run(newId(), id, turns.length, q.track, q.question, q.hint, JSON.stringify(q.options), q.allow_free ? 1 : 0, q.multi ? 1 : 0, JSON.stringify(ledger), used, now());
    99	  d.prepare('UPDATE step1_sessions SET tracks = ?, ambiguity = ? WHERE id = ?').run(JSON.stringify(ledger), overall(ledger), id);
   100	}

   106	  db().prepare('INSERT INTO step1_sessions (id, prompt, status, ambiguity, tracks, created_at) VALUES (?, ?, ?, ?, ?, ?)').run(id, prompt, 'interviewing', overall(ledger), JSON.stringify(ledger), now());
   107	  await askNext(id);
   108	  return id;
   109	}

   113	export async function act(id: string, body: Record<string, unknown>): Promise<State | null> {
   114	  const x = rows(id);
   115	  if (!x) return null;
   116	  const open = x.t.find((r) => !r.answer);
   117	  const d = db();
   118	  switch (body.action) {
   119	    case 'answer': {
   120	      if (x.s.status !== 'interviewing' || !open || body.turnId !== open.id) throw new InputError('이미 답한 질문이에요. 화면을 새로 고쳐 주세요.');
   121	      const options: string[] = JSON.parse(open.options);
   122	      const raw = Array.isArray(body.choices) ? body.choices : [];
   123	      const choices = [...new Set(raw.filter((c): c is string => typeof c === 'string'))];
   124	      if (choices.some((c) => c !== RECOMMEND && !options.includes(c))) throw new InputError('보기에 없는 답이 섞여 있어요.');
   125	      if (!open.multi && choices.length > 1) throw new InputError('이 질문은 하나만 고를 수 있어요.');
   126	      const text = typeof body.text === 'string' ? body.text.trim() : '';
   127	      if (text.length > 500) throw new InputError('직접 쓰기는 500자까지예요.');
   128	      if (!open.allow_free && text) throw new InputError('이 질문은 직접 쓰기를 받지 않아요.');
   129	      if (!choices.length && !text) throw new InputError('보기를 고르거나 직접 써 주세요.');
   130	      const answer: Answer = { choices, text };
   131	      const ledger = applyAnswer(x.ledger, open.track, answer);
   132	      d.prepare('UPDATE step1_turns SET answer = ? WHERE id = ?').run(JSON.stringify(answer), open.id);
   133	      d.prepare('UPDATE step1_sessions SET tracks = ?, ambiguity = ? WHERE id = ?').run(JSON.stringify(ledger), overall(ledger), id);
   134	      await askNext(id);
   135	      break;
   136	    }
   137	    case 'wrap': // 여기까지로 정리하기: drop the open question, go to the Restate gate
   138	      if (x.s.status !== 'interviewing') break;
   139	      if (!x.t.some((r) => r.answer)) throw new InputError('질문에 하나 이상 답해야 정리할 수 있어요.');
   140	      if (open) d.prepare('DELETE FROM step1_turns WHERE id = ?').run(open.id);
   141	      break;
   142	    case 'more': // Restate gate: "질문 하나 더"
   143	      if (x.s.status === 'interviewing' && !open) await askNext(id, true);
   144	      break;
   145	    case 'back': { // 이전: drop the open question, reopen the last answered one and restore its ledger
   146	      if (x.s.status !== 'interviewing') break;
   147	      const last = [...x.t].reverse().find((r) => r.answer);
   148	      if (!last) break;
   149	      if (open) d.prepare('DELETE FROM step1_turns WHERE id = ?').run(open.id);
   150	      d.prepare('UPDATE step1_turns SET answer = NULL WHERE id = ?').run(last.id);
   151	      const ledger = last.ledger ?? x.s.tracks ?? '{}';
   152	      d.prepare('UPDATE step1_sessions SET tracks = ?, ambiguity = ? WHERE id = ?').run(ledger, overall(JSON.parse(ledger)), id);
   153	      break;
   154	    }
   155	    case 'finish': { // Restate gate confirmed: write the ELI5 concept
   156	      if (x.s.status !== 'interviewing' || open) throw new InputError('정리할 준비가 아직 안 됐어요.');
   157	      const turns = x.t.map(toTurn);
   158	      if (!turns.length) throw new InputError('질문에 하나 이상 답해야 정리할 수 있어요.');
   159	      const fallback = mockConcept(x.s.prompt, turns);
   160	      const res = await completeJSON(CONCEPT_SYSTEM, conceptUser(x.s.prompt, turns), () => fallback);
   161	      const c = parseConcept(res.data, fallback);
   162	      const concept = { what: c.what, analogy: c.analogy, details: c.details, so_what: c.so_what, decisions: decisions(turns).map(({ track, decision }) => ({ track, decision })) };
   163	      if (rows(id)!.s.status === 'done') break; // a parallel request already finished it
   164	      d.prepare('INSERT OR REPLACE INTO step1_results (session_id, title, one_liner, concept, provider, created_at) VALUES (?, ?, ?, ?, ?, ?)')
   165	        .run(id, c.title, c.one_liner, JSON.stringify(concept), res.error ? 'mock (LLM 호출 실패)' : res.provider, now());
   166	      d.prepare("UPDATE step1_sessions SET status = 'done' WHERE id = ?").run(id);
   167	      break;
   168	    }
   169	    default:
   170	      throw new InputError('알 수 없는 요청이에요.');
   171	  }
   172	  return getState(id);
   173	}

    44	function seed() {
    45	  const d = db();
    46	  if (d.prepare('SELECT 1 FROM step1_sessions LIMIT 1').get()) return;
    47	  const now = new Date().toISOString();
    48	  let ledger = initialLedger(SAMPLE_PROMPT);
