     4	/** MIME type from magic bytes; only png/jpeg/gif/webp. SVG and everything else → null. */
     5	export function sniffImage(b: Uint8Array): string | null {
     6	  const ascii = (from: number, to: number) => String.fromCharCode(...b.subarray(from, to));
     7	  if (b.length >= 8 && [0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a].every((v, i) => b[i] === v)) return 'image/png';
     8	  if (b.length >= 3 && b[0] === 0xff && b[1] === 0xd8 && b[2] === 0xff) return 'image/jpeg';
     9	  if (b.length >= 6 && (ascii(0, 6) === 'GIF87a' || ascii(0, 6) === 'GIF89a')) return 'image/gif';
    10	  if (b.length >= 12 && ascii(0, 4) === 'RIFF' && ascii(8, 12) === 'WEBP') return 'image/webp';
    11	  return null;
    12	}

   157	/** Section names missing from a Markdown document (H1 `… 기획서` counts as `title`). */
   158	export function missingSections(markdown: string): string[] {
   159	  const heads = markdown.split('\n').filter((l) => /^##\s/.test(l)).map((l) => l.replace(/^##\s+/, ''));
   160	  const missing: string[] = SECTIONS.filter((s) => !heads.some((h) => h.includes(s)));
   161	  if (!/^#\s+.+기획서\s*$/m.test(markdown)) missing.unshift('title');
   162	  return missing;
   163	}
   164	
   165	/** Make sure every image is embedded: missing ones go at the end of the 아트·분위기 section. */
   166	export function ensureImages(markdown: string, images: ImageInput[]): string {
   167	  const absent = images.filter((img) => !markdown.includes(`](${mdUrl(img.url)}`));
   168	  if (!absent.length) return markdown;
   169	  const block = absent.map(imageMd).join('\n\n');
   170	  const lines = markdown.split('\n');
   171	  const art = lines.findIndex((l) => /^##\s.*아트/.test(l));
   172	  if (art < 0) return `${markdown.trimEnd()}\n\n## 아트·분위기\n\n${block}\n`;
   173	  let end = lines.findIndex((l, i) => i > art && /^##\s/.test(l));
   174	  if (end < 0) end = lines.length;
   175	  lines.splice(end, 0, block, '');
   176	  return lines.join('\n');
   177	}
